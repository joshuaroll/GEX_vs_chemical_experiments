```python
"""
MolFormer frozen chemical encoder — SMILES → 768-dim embedding.

Model: ibm-research/MoLFormer-XL-both-10pct (HuggingFace)
HF commit SHA: 7b12d946c181a37f6012b9dc3b002275de070314
License: Apache-2.0 (IBM Research)

trust_remote_code=True is required: model uses custom tokenizer/architecture
(configuration_molformer.py, modeling_molformer.py, tokenization_molformer.py).
Security audit (2026-05-20): IBM Research publisher, Apache-2.0, standard ML files only,
no exec/eval/shell calls in model code. Approved for use.

NO fine-tuning. Model is frozen in eval mode. Do NOT call model.train() or optimizer.step().
"""
import torch
from transformers import AutoModel, AutoTokenizer

HF_MODEL_ID = "ibm-research/MoLFormer-XL-both-10pct"
HF_MODEL_SHA = "7b12d946c181a37f6012b9dc3b002275de070314"
EMBED_DIM = 768


class MolFormerWrapper:
    """Frozen MolFormer encoder. Converts SMILES strings to 768-dim embeddings."""

    def __init__(self, device: torch.device, cache_dir: str = None):
        """Load MolFormer from HuggingFace hub (or local cache).

        Args:
            device: torch.device to run inference on.
            cache_dir: Optional HF cache directory override.
        """
        self.device = device
        kwargs = dict(trust_remote_code=True)
        if cache_dir:
            kwargs["cache_dir"] = cache_dir

        self.tokenizer = AutoTokenizer.from_pretrained(HF_MODEL_ID, **kwargs)
        self.model = AutoModel.from_pretrained(
            HF_MODEL_ID,
            deterministic_eval=True,
            **kwargs,
        ).to(device)
        self.model.eval()
        # Freeze all parameters — no gradients needed
        for param in self.model.parameters():
            param.requires_grad = False

    @torch.no_grad()
    def encode(self, smiles_list: list[str], batch_size: int = 64) -> torch.Tensor:
        """Encode a list of SMILES strings to a [N, 768] embedding tensor.

        Args:
            smiles_list: List of canonical SMILES strings.
            batch_size: Batch size for HF tokenizer/model forward pass.

        Returns:
            Tensor of shape [len(smiles_list), 768] on CPU.
        """
        all_embeddings = []
        for i in range(0, len(smiles_list), batch_size):
            batch = smiles_list[i : i + batch_size]
            inputs = self.tokenizer(
                batch,
                return_tensors="pt",
                padding=True,
                truncation=True,
                max_length=202,  # MolFormer max_position_embeddings
            )
            inputs = {k: v.to(self.device) for k, v in inputs.items()}
            outputs = self.model(**inputs)
            # Use pooler_output (768-dim CLS embedding) if available
            if outputs.pooler_output is not None:
                emb = outputs.pooler_output  # [batch, 768]
            else:
                emb = outputs.last_hidden_state[:, 0, :]  # CLS token fallback
            all_embeddings.append(emb.cpu())
        return torch.cat(all_embeddings, dim=0)  # [N, 768]


def load_molformer(device: torch.device, cache_dir: str = None) -> MolFormerWrapper:
    """Convenience factory. Returns a MolFormerWrapper ready for encode()."""
    return MolFormerWrapper(device=device, cache_dir=cache_dir)
```

```python
def load_multidcp_model(ckpt_path: str, device: torch.device):
    """Load a MultiDCP_AE checkpoint. Returns (model, model_params)."""
    sys.path.insert(0, str(MULTIDCP_ROOT))
```

```python
def build_landmark_filter():
    """Load landmark gene list and overlap with diseased baseline columns.

    Returns:
        landmark_genes: list of gene names (the overlap set)
        gene_vectors: np.array [num_gene, 128] (gene topology for MultiDCP)
        baseline_per_cell: dict {cell_id: np.array [num_gene]} (diseased expression)
    """
    gene_df = pd.read_csv(GENE_VECTOR_FILE, index_col=0)
    # gene_df index = gene names, columns = 128 embedding dims
    all_landmark = gene_df.index.tolist()

    diseased_df = pd.read_csv(DISEASED_FILE, index_col=0)
    diseased_cols = set(diseased_df.columns.tolist())

    # Intersection — ONLY use genes present in both landmark list AND diseased file
    landmark_genes = [g for g in all_landmark if g in diseased_cols]
    # Note: design doc says 978; actual overlap is 918 (confirmed 2026-05-20)
    print(f'Landmark gene overlap: {len(landmark_genes)} genes '
          f'(gene_vector.csv had {len(all_landmark)}, diseased file has {len(diseased_cols)} unique cols)')

    gene_vectors = gene_df.loc[landmark_genes].values  # [num_gene, 128]

    baseline_per_cell = {
        cell: diseased_df.loc[cell, landmark_genes].values.astype(np.float32)
        for cell in CELLS_10
    }
    return landmark_genes, gene_vectors, baseline_per_cell
```

```python
@torch.no_grad()
def infer_drug_cell(model, drug_feature, cell_baseline_tensor,
                    gene_vectors_tensor, device, is_gex: bool):
    """Run one drug-cell forward pass. Returns np.array (scalar for DOSE, [num_gene] for GEX).

    Args:
        model: MultiDCP_AE (eval mode, frozen)
        drug_feature: drug graph/fingerprint (np.array or DGLGraph — whatever drug_feature_extract returns)
        cell_baseline_tensor: torch.FloatTensor [num_gene] — diseased cell expression
        gene_vectors_tensor: torch.FloatTensor [num_gene, 128] — gene topology
        device: torch device
        is_gex: if True, return perturbed GEX DE (subtract baseline); if False, return E-Hill scalar
    """
    # Construct the feature dict matching MultiDCP_AE forward signature
    # (replicating data_utils.transform_to_tensor_per_dataset per-row logic)
    drug_t = torch.tensor(drug_feature, dtype=torch.float32).unsqueeze(0).to(device)  # [1, drug_dim]
    cell_t = cell_baseline_tensor.unsqueeze(0).to(device)  # [1, num_gene]
    gene_t = gene_vectors_tensor.unsqueeze(0).to(device)   # [1, num_gene, 128]

    # pert_idose: all training data used 'x' (dose-aggregated) → single level → zeros one-hot
    # pert_idose_input_dim = 1 at training → pass zeros (or skip if model ignores it)
    # Most MultiDCP_AE forward signatures: model(drug, cell_id, gene, pert_type=None, pert_idose=None)
    # Check actual forward signature in multidcp_balanceloss.py and call accordingly

    out = model(drug_t, cell_t, gene_t, job_id='perturbed')
    # out shape: [1, num_gene] for GEX, or [1] / scalar for DOSE
    out_np = out.squeeze().cpu().numpy()

    if is_gex:
        # DE rule: DE = predicted_perturbed - diseased_baseline
        baseline_np = cell_baseline_tensor.cpu().numpy()
        return out_np - baseline_np  # [num_gene]
    else:
        # E-Hill scalar — return as float
        return float(out_np) if out_np.ndim == 0 else float(out_np[0])
```

```python
def main():
    args = parse_args()
    device = torch.device('cuda:0' if torch.cuda.is_available() else 'cpu')
    print(f'Using device: {device}')

    if not args.no_wandb:
        wandb.init(project=args.wandb_project, group=args.wandb_group,
                   name=args.wandb_name, config=vars(args))

    # Load landmark filter and cell baselines
    landmark_genes, gene_vectors_np, baseline_per_cell = build_landmark_filter()
    num_gene = len(landmark_genes)
    gene_vectors_tensor = torch.tensor(gene_vectors_np, dtype=torch.float32)

    # Load models
    print('Loading MODEL_DOSE...')
    model_dose, dose_params = load_multidcp_model(args.dose_ckpt, device)
    print(f'  num_gene from chkpt_dose: {dose_params["num_gene"]}')

    print('Loading MODEL_GEX...')
    model_gex, gex_params = load_multidcp_model(args.gex_ckpt, device)
    gex_num_gene = gex_params['num_gene']
    print(f'  num_gene from chkpt_gex: {gex_num_gene}')

    # Sanity check: num_gene must match landmark_genes
    assert gex_num_gene == num_gene, (
        f'num_gene mismatch: checkpoint has {gex_num_gene}, '
        f'landmark filter found {num_gene}. Check gene_vector.csv alignment.')

    print('Loading MolFormer...')
    molformer = load_molformer(device=device)

    # Load drug features
    drug_feature_dict = load_drug_features()

    # Load DILIst canonical
    dili_df = pd.read_csv(args.dili_csv)
    assert len(dili_df) == 1118, f'Expected 1118 drugs, got {len(dili_df)}'
    print(f'DILIst drugs: {len(dili_df)}')

    # Pre-encode all MolFormer embeddings (batched)
    print('Encoding MolFormer embeddings for all 1118 drugs...')
    smiles_list = dili_df['canonical_smiles'].tolist()
    molformer_embeddings = molformer.encode(smiles_list, batch_size=args.batch_size)
    # molformer_embeddings: [1118, 768] tensor on CPU

    # Per-cell baseline tensors
    baseline_tensors = {
        cell: torch.tensor(vec, dtype=torch.float32) for cell, vec in baseline_per_cell.items()
    }

    # Main feature extraction loop
    rows = []
    for drug_idx, row in dili_df.iterrows():
        drug_name = row['drug_name']
        canonical_smiles = row['canonical_smiles']

        # Get drug feature (MultiDCP drug graph)
        drug_feat = drug_feature_dict.get(drug_name.lower())
        if drug_feat is None:
            # Try SMILES-based lookup (fallback)
            drug_feat = drug_feature_dict.get(canonical_smiles)
        if drug_feat is None:
            print(f'WARNING: drug {drug_name} not found in drug_feature_dict — using zero vector')
            drug_feat = np.zeros(drug_feature_dict[next(iter(drug_feature_dict))].shape)

        # Per-cell inference
        dose_preds = []  # list of scalars
        gex_preds  = []  # list of [num_gene] arrays (DE)

        for cell in CELLS_10:
            cell_baseline = baseline_tensors[cell]

            # MODEL_DOSE inference
            dose_val = infer_drug_cell(
                model_dose, drug_feat, cell_baseline, gene_vectors_tensor,
                device=device, is_gex=False,
            )
            dose_preds.append(dose_val)

            # MODEL_GEX inference (DE = predicted_perturbed - baseline)
            gex_de = infer_drug_cell(
                model_gex, drug_feat, cell_baseline, gene_vectors_tensor,
                device=device, is_gex=True,
            )
            gex_preds.append(gex_de)

        # Mean-pool over 10 cells
        feat_dose = np.mean(dose_preds)                  # scalar
        feat_gex  = np.mean(np.stack(gex_preds), axis=0) # [num_gene]
        feat_embed = molformer_embeddings[drug_idx].numpy() # [768]

        # Build output row
        out_row = {
            'drug_name':    drug_name,
            'pert_id':      row['pert_id'],
            'dili_binary':  row['dili_binary'],
            'dili_severity': row['dili_severity'],
            'in_lincs':     row['in_lincs'],
            'in_pdg':       row['in_pdg'],
            'feat_dose':    feat_dose,
        }
        for j, val in enumerate(feat_gex):
            out_row[f'feat_gex_{j}'] = val
        for j, val in enumerate(feat_embed):
            out_row[f'feat_embed_{j}'] = val
        rows.append(out_row)

        if (drug_idx + 1) % 100 == 0:
            print(f'  Processed {drug_idx + 1}/1118 drugs')
            if not args.no_wandb:
                wandb.log({'progress/drugs_done': drug_idx + 1})

    # Write parquet
    out_df = pd.DataFrame(rows)
    Path(args.out).parent.mkdir(parents=True, exist_ok=True)
    out_df.to_parquet(args.out, index=False)
    print(f'Wrote {args.out}: shape {out_df.shape}')

    if not args.no_wandb:
        wandb.log({
            'output/n_drugs': len(out_df),
            'output/n_cols': len(out_df.columns),
            'output/n_nan': int(out_df.isnull().sum().sum()),
            'output/feat_dose_mean': float(out_df['feat_dose'].mean()),
            'output/feat_dose_std':  float(out_df['feat_dose'].std()),
        })
        wandb.finish()

    print('Phase 3 Plan 1 complete.')


if __name__ == '__main__':
    main()
```
