"""
Regression tests for Stage-2 feature parquet produced by Phase 3.

These tests are sanity checks only — they validate shape, NaN, and label
consistency. They do NOT test model outputs (values will shift across seeds).

Run:
  conda run -n dili_v04_env pytest tests/test_stage2_features.py -v
"""
import os
import pytest
import pandas as pd
import numpy as np

FEATURES_PATH = os.path.join(
    os.path.dirname(__file__), '..', 'data', 'processed', 'dili_features.parquet'
)
EXPECTED_DRUGS = 1118
EXPECTED_EMBED_DIM = 768
EXPECTED_DOSE_COLS = 1


@pytest.fixture(scope='module')
def features_df():
    """Load dili_features.parquet once for all tests in this module."""
    if not os.path.exists(FEATURES_PATH):
        pytest.skip(f'dili_features.parquet not found at {FEATURES_PATH} — run Phase 3 first')
    return pd.read_parquet(FEATURES_PATH)


def test_row_count(features_df):
    """Parquet must have exactly 1,118 rows (one per DILIst drug)."""
    assert len(features_df) == EXPECTED_DRUGS, (
        f'Expected {EXPECTED_DRUGS} rows, got {len(features_df)}'
    )


def test_no_nan(features_df):
    """Zero NaN values in any feature column (EMBED-04 hard gate)."""
    nan_count = features_df.isnull().sum().sum()
    assert nan_count == 0, (
        f'Found {nan_count} NaN values. '
        f'Affected columns: {features_df.columns[features_df.isnull().any()].tolist()[:5]}'
    )


def test_feat_dose_exists(features_df):
    """feat_dose column must exist as a single scalar column."""
    assert 'feat_dose' in features_df.columns
    dose_cols = [c for c in features_df.columns if c.startswith('feat_dose')]
    assert len(dose_cols) == EXPECTED_DOSE_COLS, (
        f'Expected 1 feat_dose column, got {len(dose_cols)}'
    )


def test_feat_embed_dim(features_df):
    """MolFormer embedding must be exactly 768-dim (feat_embed_0 through feat_embed_767)."""
    embed_cols = [c for c in features_df.columns if c.startswith('feat_embed_')]
    assert len(embed_cols) == EXPECTED_EMBED_DIM, (
        f'Expected {EXPECTED_EMBED_DIM} feat_embed columns, got {len(embed_cols)}'
    )
    # Check index continuity
    embed_indices = sorted(int(c.split('_')[-1]) for c in embed_cols)
    assert embed_indices == list(range(EXPECTED_EMBED_DIM)), 'feat_embed indices not contiguous'


def test_feat_gex_exists(features_df):
    """feat_gex columns must exist and have a reasonable dim (should be ~918)."""
    gex_cols = [c for c in features_df.columns if c.startswith('feat_gex_')]
    assert len(gex_cols) > 0, 'No feat_gex columns found'
    # Expected: ~918. Accept any value >= 100 (safety margin).
    assert len(gex_cols) >= 100, f'feat_gex dim suspiciously small: {len(gex_cols)}'
    # Check index continuity
    gex_indices = sorted(int(c.split('_')[-1]) for c in gex_cols)
    assert gex_indices == list(range(len(gex_cols))), 'feat_gex indices not contiguous'


def test_label_distribution(features_df):
    """dili_binary must have both 0 and 1 labels with the expected counts."""
    assert 'dili_binary' in features_df.columns
    vc = features_df['dili_binary'].value_counts().to_dict()
    assert 1 in vc and 0 in vc, f'Missing label class. Got: {vc}'
    assert vc[1] == 685, f'Expected 685 DILI-positive, got {vc[1]}'
    assert vc[0] == 433, f'Expected 433 DILI-negative, got {vc[0]}'


def test_feat_dose_finite(features_df):
    """feat_dose must be finite (no Inf/-Inf)."""
    assert np.isfinite(features_df['feat_dose'].values).all(), 'feat_dose contains Inf values'


def test_feat_embed_finite(features_df):
    """MolFormer embeddings must all be finite."""
    embed_cols = [c for c in features_df.columns if c.startswith('feat_embed_')]
    mat = features_df[embed_cols].values
    assert np.isfinite(mat).all(), 'feat_embed contains Inf/NaN values'


def test_metadata_columns_present(features_df):
    """Required metadata columns must be present."""
    required = ['drug_name', 'pert_id', 'dili_binary', 'dili_severity', 'in_lincs', 'in_pdg']
    for col in required:
        assert col in features_df.columns, f'Missing column: {col}'
