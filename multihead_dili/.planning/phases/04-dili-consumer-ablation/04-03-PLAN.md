### Task 1: HG3 — embed-only (var1) random-split AUROC check

```python
import pandas as pd, glob, numpy as np
from sklearn.metrics import roc_auc_score

files = glob.glob('data/processed/predictions/var1_head*_splitrandom_*.parquet')
aurocs = []
for f in files:
    df = pd.read_parquet(f)
    if len(df['true_label'].unique()) >= 2:
        aurocs.append(roc_auc_score(df['true_label'], df['predicted_proba']))

mean_auroc = np.mean(aurocs)
hg3_pass = mean_auroc >= 0.55
print(f'embed-only random AUROC: mean={mean_auroc:.4f} std={np.std(aurocs):.4f} (n={len(aurocs)})')
print(f'HG3: {"PASS" if hg3_pass else "HALT"}')
```
